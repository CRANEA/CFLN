function doPost(e) { try { // 1. BACKEND ANTI-SPAM CHECK (The Bouncer) // If the hidden answer isn't exactly '9', drop the request completely. if (e.parameter.spamAnswer !== '9') { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": "Spam blocked" })).setMimeType(ContentService.MimeType.JSON); } // 2. OPEN THE GOOGLE SHEET var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 3. GET THE DATA FROM THE FORM var timestamp = new Date(); var name = e.parameter.name || "No Name"; var email = e.parameter.email || "No Email"; var phone = e.parameter.phone || "No Phone"; var concerns = e.parameter.concerns || "None"; // 4. ADD ROW TO THE SHEET sheet.appendRow([timestamp, name, email, phone, concerns]); // 5. SEND THE EMAIL NOTIFICATION var adminEmail = "avi@craneandassociates.net"; var subject = "New Fiduciary Request from " + name; var body = "You have a new request from the website.\n\n" + "Name: " + name + "\n" + "Email: " + email + "\n" + "Phone: " + phone + "\n" + "Concerns: " + concerns; MailApp.sendEmail({ to: adminEmail, subject: subject, body: body }); // 6. RETURN SUCCESS TO THE WEBSITE return ContentService.createTextOutput(JSON.stringify({ "result": "success" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
